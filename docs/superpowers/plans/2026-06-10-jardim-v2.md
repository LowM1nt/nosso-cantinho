# Jardim dos Laços v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Modernizar o visual (glassmorphism rosa) e tornar as fases de quinta/sexta/sábado multi-etapa e bem mais difíceis, com dicas progressivas e fragmento via passo bônus.

**Architecture:** Um "motor de fases" data-driven (`phaseEngine`) roda fases multi-etapa com barra de progresso, dicas que destravam por tempo+tentativas, e um passo bônus que coleta o fragmento. Conteúdo (textos/respostas, romântico+engraçado) fica centralizado em `js/content.js`. Lógica pura (gating de dicas, comparação de respostas, progresso) é testável com `node --test`. Visual é Tailwind (CDN) + classes `.glass` em `css/styles.css`.

**Tech Stack:** HTML + Tailwind CDN + ES modules (vanilla JS), testes com `node:test`. Sem build. Rodar testes: `npm test` (precisa `export PATH="$PATH:/c/Program Files/nodejs"`).

`js/engine/hints.js`:
```js
// Quanto destravar de dica: por tentativas OU tempo real (o que vier primeiro).
export const DEFAULT_THRESHOLDS = [
  { errors: 3, ms: 240000 },   // nível 1: 3 erros ou 4 min
  { errors: 6, ms: 480000 },   // nível 2: 6 erros ou 8 min
  { errors: 10, ms: 900000 },  // nível 3 (quase-resposta): 10 erros ou 15 min
];

export function unlockedHintLevel(attempts, elapsedMs, thresholds = DEFAULT_THRESHOLDS) {
  let level = 0;
  for (let i = 0; i < thresholds.length; i++) {
    if (attempts >= thresholds[i].errors || elapsedMs >= thresholds[i].ms) level = i + 1;
  }
  return level;
}
```

import { normalizeText } from '../normalize.js';

const norm = s => normalizeText(String(s ?? '')).replace(/\s/g, '');

// Aceita uma resposta (string) ou várias (array). Ignora acento/caixa/espaço.
export function answerMatches(input, answer) {
  const got = norm(input);
  if (!got) return false;
  const list = Array.isArray(answer) ? answer : [answer];
  return list.some(a => got === norm(a));
}

## Task 6: Phase engine (controller)

**Files:**
- Create: `js/engine/phaseEngine.js`

Engine roda uma lista de etapas dentro do `#enigma-body`. Cada etapa: `{ titulo, render(host, ctx) }`.
`ctx` oferece: `ctx.solved()` (avança/finaliza), `ctx.wrong()` (conta erro + atualiza dica), `ctx.now()` (ms desde início da etapa), `ctx.say(txt)` (mensagem), `ctx.hintButton(hints)` (mostra botão de dica que revela `hints[nível-1]`). Helper `textStep(cfg)` cria etapa de input de texto padrão.

- [ ] **Step 1: Implement engine + textStep**

`js/engine/phaseEngine.js`:
```js
import { unlockedHintLevel } from './hints.js';
import { answerMatches } from './answer.js';
import { getPhaseStep, setPhaseStep } from '../state.js';

// steps: [{ titulo, render(host, ctx) }]
// bonus: { titulo, render(host, ctx) }  -> ao resolver, chama onCollect()
// callbacks: { onSolvedAll, onCollect, onWrong }
export function runPhase(host, { fase, steps, bonus, state, persist, doc = document,
                                 onCollect, onWrong }) {
  let idx = Math.min(getPhaseStep(state, fase), steps.length); // retoma
  const all = bonus ? [...steps, { ...bonus, _bonus: true }] : steps;

  function drawProgress() {
    const dots = all.map((_, i) =>
      `<span class="step-dot ${i < idx ? 'done' : i === idx ? 'current' : ''}"></span>`).join('');
    return `<div class="step-dots mb-3">${dots}</div>`;
  }

  function renderStep() {
    const step = all[idx];
    host.innerHTML = `${drawProgress()}
      <div class="text-center text-sm mb-2 opacity-70">${step._bonus ? '🎁 Baú secreto' : `Etapa ${idx + 1}/${steps.length}`}</div>
      <h3 class="font-titulo text-cereja text-center mb-3">${step.titulo}</h3>
      <div id="step-host" class="anim-fadeup"></div>
      <div id="hint-zone" class="text-center mt-3"></div>
      <p id="step-msg" class="text-center text-cereja mt-2 h-6"></p>`;

    const stepHost = host.querySelector('#step-host');
    const hintZone = host.querySelector('#hint-zone');
    const msgEl = host.querySelector('#step-msg');
    let attempts = 0;
    const startedAt = Date.now();

    const ctx = {
      say(t) { msgEl.textContent = t; },
      wrong() { attempts++; refreshHint(); if (onWrong) onWrong(); },
      hintButton(hints) {
        ctx._hints = hints;
        refreshHint();
      },
      solved() {
        if (step._bonus) { onCollect(); return; }
        idx++;
        Object.assign(state, setPhaseStep(state, fase, idx));
        persist();
        if (idx >= all.length) { onCollect(); return; } // sem bônus: última etapa coleta
        renderStep();
      },
    };

    function refreshHint() {
      if (!ctx._hints) return;
      const lvl = unlockedHintLevel(attempts, Date.now() - startedAt);
      if (lvl <= 0) { hintZone.innerHTML = ''; return; }
      const maxLvl = Math.min(lvl, ctx._hints.length);
      hintZone.innerHTML = `<button id="hint-btn" class="text-xs underline opacity-80">🎀 pedir dicinha (${maxLvl})</button>
        <p id="hint-text" class="text-xs mt-1 opacity-80"></p>`;
      hintZone.querySelector('#hint-btn').addEventListener('click', () => {
        hintZone.querySelector('#hint-text').textContent = ctx._hints[maxLvl - 1];
      });
    }
    // re-checa dica por tempo a cada 20s
    const timer = setInterval(refreshHint, 20000);
    ctx._cleanup = () => clearInterval(timer);

    step.render(stepHost, ctx);
  }

  renderStep();
}

// Etapa de texto padrão. cfg: { prompt, resposta (string|array), hints:[], placeholder, decorHTML }
export function textStep({ titulo, prompt, resposta, hints = [], placeholder = 'resposta', decorHTML = '' }) {
  return {
    titulo,
    render(host, ctx) {
      host.innerHTML = `${decorHTML}
        <p class="mb-2">${prompt}</p>
        <input id="ts-input" class="w-full rounded-full border-2 border-rosa px-4 py-2 text-center" placeholder="${placeholder}" />
        <button id="ts-btn" class="mt-3 w-full rounded-full bg-cereja text-marfim font-titulo py-2 btn-glow">Conferir</button>`;
      ctx.hintButton(hints);
      const input = host.querySelector('#ts-input');
      const go = () => {
        if (answerMatches(input.value, resposta)) return ctx.solved();
        ctx.wrong(); ctx.say('Hmm, não foi dessa vez 🎀');
      };
      host.querySelector('#ts-btn').addEventListener('click', go);
      input.addEventListener('keydown', e => { if (e.key === 'Enter') go(); });
    },
  };
}
```
